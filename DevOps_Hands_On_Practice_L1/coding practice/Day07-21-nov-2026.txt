##  # Ansible - Install Nginx

---
- name: install nginx
  hosts: all
  become: yes

  task:
    - name: Install Nginx
      yum:
        name: Nginx
        state: present

    - name: enable and start nginx
      yum:
        name: nginx
        enable: yes
        state: started

##  # terraform - create EC2

##  # Kubernetes - Deploy 2 Replicas

apiVersion: apps/v1
kind: Deployment
metadata:
  name: pathnex-deployment
  labels:
    app: pathnex
spec:
  replicas: 2
  selector:
    matchLabels:
      app: pathnex-app
  template:
    metadata:
      labels:
        app: pathnex-app
    spec:
      containers:
      - name: pathnex-container
        image: nginx:latest
        ports:
        - containerPort: 80

##  # Jenkins Pipeline - Parameterized
you will learn how to **create parameterized build in jenkins**

pipeline {
    agent any
    parameters {
        string(name: 'ENVIRONMENT', defaultValue: 'dev', description: 'Deployment Environment')

    }
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    stages {
        stage('dheckout') {
            steps {
                git url: 'https://github.com/Pathnex/sample-java-app.git'
            }
        }
        stage('Deploy') {
            steps {
                sh 'echo "Deplying to $Environment by $INSTITUTE_NAME"'
            }
        }
    }
}
