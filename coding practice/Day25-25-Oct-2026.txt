# Terraform - Create CloudWatch alarms
resource "aws_cloudwatch_metric_alarm" "pathnex_cpu_alarm" {
  alarm_name            = "pathnex-cpu-utilization-high"
  comparison_operator   = "GreaterThanThreshold"
  evaluation_periods    = "1"
  metric_name           = "CPUUtilization"
  namespace             = "AWS/EC2"
  period                = "60"
  statestic             = "average"
  threshold             = "80"

  dimensions = {
    InstanceId = aws_instance.pathnex_ec2.id
}

alarm_actions = [aws_sns_topic.pathnex_alarts.arn]
}

resource "aws_sns_topic" "pathnex_alerts" {
  name = "pathnex-alarts"
}

# Jenkinsfile - Continuous Monitering and alert in CI pipeline
pipeline {
    agent any
    stages {
        stage('monitor') {
            steps {
                script {
                    echo 'Triggering CloudWatch if CPU exceeds 80%'
                    sh 'aws cloudwatch put-metric-data --namespace "AWS/EC2 --METRIC-NAME "CPUUtilization" --value 90'
                }
            }
        }
        stage('Test') {
            steps {
                echo 'Running tests...'
            }
        }
    }
}
