Day 09 - Docker Integration with Jenkins and Gitlab
# Ansible - Setup Docker Container for Nginx
- name: Setup Docker Container for Nginx
  host: all
  become: yes
  tasks:
    - name: Pull Nginx image
      docker_ image:
        name: nginx
        source: pull
    - name: Run Nginx container
      docker_container:
        name: nginx-container
        image: nginx
        state: started
        published_ports:
            - "8080:80"
