## # Jenkinsfile - with Git Integration and Testing

```groovy
pipeline{
    agent any
    stages {
        stage('Clone Repository') {
            steps {
                git 'https://github.com/pathnex/sample-repo.git'
            }
        }
        stage('Build') {
            steps {
                echo 'Building project from Git repository'
            }
        }
        stage('Test') {
            steps {
                echo 'Running unit tests'
            }
        }
        stage('Deploy') {
            steps {
                echo 'Deploying application'
            }
        }
    }
}

# Ansible - Create user and Set Permissions
- name: Creaete a new uset
    hosts: all
    become: yes
    tasks:
      - name: Creaete user
      user:
        name: pathnex
        state: present
        shell: /bin/bash
    - name: Set permissions for the User
    file:
      path: /home/pathnex
      owner: pathnex
      group: pathnex
      mode: '0755'
